# Keyboard Shortcuts

main + Q = terminal (alacritty)
main + M = disables/enables main screen when dual monitoring
main + N = opens Obsidian
main + R = opens wofi (Application Launcher)
main + C = close
main + B = opens Brave

printscreen = copy to clipboard print screen click whole monitor

# Workspaces Shortcut

main + 1...0 = change workspace
